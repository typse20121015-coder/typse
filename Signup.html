/** * 石門福華報名系統 - Google Apps Script 後端 API * 支援：讀取 (GET)、新增 (POST: add)、修改 (POST: update)、刪除 (POST: delete) */ // 取得預設的第一張工作表 function getSheet() { return SpreadsheetApp.getActiveSpreadsheet().getSheets()[0]; } // 供管理員初始建立表頭使用的輔助函式 function setup() { var sheet = getSheet(); if (sheet.getLastRow() === 0) { sheet.appendRow(['時間戳記', '系統ID (請勿修改)', '會員姓名', '報名總人數', '詳細資料JSON (請勿修改)']); sheet.getRange("A1:E1").setFontWeight("bold").setBackground("#d0e0e3"); // 凍結第一列 sheet.setFrozenRows(1); // 調整欄寬以利閱讀 sheet.setColumnWidth(1, 180); sheet.setColumnWidth(2, 120); sheet.setColumnWidth(5, 400); } } // 處理前端的讀取請求 (取得所有報名資料) function doGet(e) { var sheet = getSheet(); var data = sheet.getDataRange().getValues(); var result = []; // 從第二列開始讀取 (略過表頭) for (var i = 1; i < data.length; i++) { // JSON 資料儲存在第 5 欄 (陣列索引 4) if (data[i][4]) { try { var obj = JSON.parse(data[i][4]); obj.timestamp = data[i][0]; obj.userId = data[i][1]; result.push(obj); } catch (error) { // 忽略解析錯誤的無效列 } } } // 回傳 JSON 給前端的 React 應用程式 return ContentService.createTextOutput(JSON.stringify(result)) .setMimeType(ContentService.MimeType.JSON); } // 處理前端的新增、修改、刪除請求 function doPost(e) { var sheet = getSheet(); var payload; try { // 前端送來的格式是 text/plain，需要手動 parse payload = JSON.parse(e.postData.contents); } catch (err) { return ContentService.createTextOutput(JSON.stringify({ success: false, message: "無效的資料格式" })) .setMimeType(ContentService.MimeType.JSON); } var action = payload.action; var userId = payload.userId; // ---------------------------------------------------- // 1. 新增報名資料 (Add) // ---------------------------------------------------- if (action === 'add') { var timestamp = new Date().toISOString(); var memberName = payload.memberName; var totalPeople = 1 + (payload.guests ? payload.guests.length : 0); // 移除不需存入詳細 JSON 中的系統用欄位 delete payload.action; delete payload.userId; // 將資料寫入試算表的新列 sheet.appendRow([ timestamp, userId, memberName, totalPeople, JSON.stringify(payload) ]); return ContentService.createTextOutput(JSON.stringify({ success: true, message: "新增成功" })) .setMimeType(ContentService.MimeType.JSON); } // ---------------------------------------------------- // 2. 刪除報名資料 (Delete) // ---------------------------------------------------- if (action === 'delete') { var data = sheet.getDataRange().getValues(); for (var i = 1; i < data.length; i++) { if (data[i][1] === userId) { sheet.deleteRow(i + 1); // 陣列索引是 0 開始，但試算表列數從 1 開始，故要 +1 return ContentService.createTextOutput(JSON.stringify({ success: true, message: "刪除成功" })) .setMimeType(ContentService.MimeType.JSON); } } return ContentService.createTextOutput(JSON.stringify({ success: false, message: "找不到該筆資料" })) .setMimeType(ContentService.MimeType.JSON); } // ---------------------------------------------------- // 3. 更新報名資料 (Update) // ---------------------------------------------------- if (action === 'update') { var data = sheet.getDataRange().getValues(); var memberName = payload.memberName; var totalPeople = 1 + (payload.guests ? payload.guests.length : 0); delete payload.action; delete payload.userId; for (var i = 1; i < data.length; i++) { if (data[i][1] === userId) { // 更新試算表上的對應欄位 sheet.getRange(i + 1, 3).setValue(memberName); sheet.getRange(i + 1, 4).setValue(totalPeople); sheet.getRange(i + 1, 5).setValue(JSON.stringify(payload)); return ContentService.createTextOutput(JSON.stringify({ success: true, message: "更新成功" })) .setMimeType(ContentService.MimeType.JSON); } } return ContentService.createTextOutput(JSON.stringify({ success: false, message: "找不到該筆資料" })) .setMimeType(ContentService.MimeType.JSON); } // ---------------------------------------------------- // 預設錯誤回傳 // ---------------------------------------------------- return ContentService.createTextOutput(JSON.stringify({ success: false, message: "未知的操作請求" })) .setMimeType(ContentService.MimeType.JSON); }
